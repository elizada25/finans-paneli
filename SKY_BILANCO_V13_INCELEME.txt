export async function GET(request) {
  try {
    const { searchParams } = new URL(request.url);

    const symbol = String(searchParams.get('symbol') || '')
      .trim()
      .toUpperCase();

    if (!symbol) {
      return Response.json(
        { error: 'Hisse kodu gerekli.' },
        { status: 400 }
      );
    }

    const cik = await resolveCIK(symbol);

    if (!cik) {
      return Response.json(
        { error: `${symbol} için SEC kaydı bulunamadı.` },
        { status: 404 }
      );
    }

    const [factsResponse, submissionsResponse] = await Promise.all([
      fetch(
        `https://data.sec.gov/api/xbrl/companyfacts/CIK${cik}.json`,
        {
          cache: 'no-store',
          headers: {
            'User-Agent': 'Sky-Finans/1.2',
            Accept: 'application/json',
          },
        }
      ),
      fetch(
        `https://data.sec.gov/submissions/CIK${cik}.json`,
        {
          cache: 'no-store',
          headers: {
            'User-Agent': 'Sky-Finans/1.2',
            Accept: 'application/json',
          },
        }
      ),
    ]);

    if (!factsResponse.ok) {
      throw new Error('SEC finansal verileri alınamadı.');
    }

    const factsJson = await factsResponse.json();
    const submissionsJson =
      submissionsResponse.ok
        ? await submissionsResponse.json()
        : {};

    const facts = factsJson?.facts?.['us-gaap'] || {};

    const metrics = {
      revenue: getMetric(
        facts,
        [
          'RevenueFromContractWithCustomerExcludingAssessedTax',
          'Revenues',
          'SalesRevenueNet',
        ],
        ['USD']
      ),

      netIncome: getMetric(
        facts,
        ['NetIncomeLoss', 'ProfitLoss'],
        ['USD']
      ),

      eps: getMetric(
        facts,
        [
          'EarningsPerShareDiluted',
          'EarningsPerShareBasic',
        ],
        ['USD/shares']
      ),

      cash: getMetric(
        facts,
        [
          'CashAndCashEquivalentsAtCarryingValue',
          'CashCashEquivalentsRestrictedCashAndRestrictedCashEquivalents',
        ],
        ['USD']
      ),

      operatingCashFlow: getMetric(
        facts,
        [
          'NetCashProvidedByUsedInOperatingActivities',
        ],
        ['USD']
      ),

      assets: getMetric(
        facts,
        ['Assets'],
        ['USD']
      ),

      liabilities: getMetric(
        facts,
        ['Liabilities'],
        ['USD']
      ),
    };

    const filing = getLatestFiling(submissionsJson);

    const result = buildFinancialAnalysis(
      symbol,
      metrics,
      filing
    );

    return Response.json(
      result,
      {
        headers: {
          'Cache-Control': 'no-store, max-age=0',
        },
      }
    );
  } catch (error) {
    console.error('Sky financials hatası:', error);

    return Response.json(
      {
        error:
          error instanceof Error
            ? error.message
            : 'Bilanço analizi yapılamadı.',
      },
      { status: 500 }
    );
  }
}

async function resolveCIK(symbol) {
  const response = await fetch(
    'https://www.sec.gov/files/company_tickers.json',
    {
      cache: 'no-store',
      headers: {
        'User-Agent': 'Sky-Finans/1.2',
        Accept: 'application/json',
      },
    }
  );

  if (!response.ok) return null;

  const json = await response.json();

  for (const row of Object.values(json || {})) {
    if (
      String(row?.ticker || '').toUpperCase() === symbol
    ) {
      return String(row?.cik_str || '').padStart(10, '0');
    }
  }

  return null;
}

function getMetric(facts, tags, units) {
  for (const tag of tags) {
    const fact = facts?.[tag];
    if (!fact?.units) continue;

    for (const unit of units) {
      const values = fact.units?.[unit];

      if (!Array.isArray(values)) continue;

      const filtered = values
        .filter(
          (item) =>
            (item.form === '10-Q' || item.form === '10-K') &&
            Number.isFinite(Number(item.val))
        )
        .sort(
          (a, b) =>
            new Date(b.filed || b.end).getTime() -
            new Date(a.filed || a.end).getTime()
        );

      const unique = [];

      for (const item of filtered) {
        if (
          unique.some(
            (x) =>
              x.end === item.end &&
              x.form === item.form
          )
        ) {
          continue;
        }

        unique.push(item);

        if (unique.length >= 2) break;
      }

      if (unique.length) {
        return {
          current: Number(unique[0].val),
          previous:
            unique[1] ? Number(unique[1].val) : null,
          end: unique[0].end,
          form: unique[0].form,
        };
      }
    }
  }

  return null;
}

function getLatestFiling(data) {
  const recent = data?.filings?.recent;

  const forms = recent?.form || [];
  const dates = recent?.filingDate || [];

  for (let i = 0; i < forms.length; i++) {
    if (forms[i] === '10-Q' || forms[i] === '10-K') {
      return {
        form: forms[i],
        date: dates[i],
      };
    }
  }

  return null;
}

function buildFinancialAnalysis(symbol, metrics, filing) {
  let score = 0;
  const comments = [];

  const revenueGrowth = growth(metrics.revenue);

  if (Number.isFinite(revenueGrowth)) {
    if (revenueGrowth >= 10) {
      score += 2;
      comments.push(
        `Gelir önceki raporlanan döneme göre %${revenueGrowth.toFixed(1)} arttı.`
      );
    } else if (revenueGrowth >= 0) {
      score += 1;
      comments.push(
        `Gelir %${revenueGrowth.toFixed(1)} arttı.`
      );
    } else {
      score -= 1;
      comments.push(
        `Gelir %${Math.abs(revenueGrowth).toFixed(1)} geriledi.`
      );
    }
  }

  const netIncome = metrics.netIncome?.current;

  if (Number.isFinite(netIncome)) {
    if (netIncome > 0) {
      score += 1;
      comments.push('Net sonuç pozitif.');
    } else {
      score -= 1;
      comments.push('Şirket raporlanan dönemde net zarar açıkladı.');
    }
  }

  const cashGrowth = growth(metrics.cash);

  if (Number.isFinite(cashGrowth)) {
    if (cashGrowth > 10) {
      score += 1;
      comments.push('Nakit pozisyonunda güçlü artış var.');
    } else if (cashGrowth < -10) {
      score -= 1;
      comments.push('Nakit pozisyonunda belirgin düşüş var.');
    }
  }

  const opCash = metrics.operatingCashFlow?.current;

  if (Number.isFinite(opCash)) {
    if (opCash > 0) {
      score += 1;
      comments.push('Operasyonel nakit akışı pozitif.');
    } else {
      score -= 1;
      comments.push('Operasyonel nakit akışı negatif.');
    }
  }

  let verdict = 'NÖTR';

  if (score >= 3) verdict = 'OLUMLU';
  else if (score <= -2) verdict = 'RİSKLİ';

  const answer = `${symbol} BİLANÇO ANALİZİ — ${verdict}

Son rapor: ${filing?.form || '—'} • ${formatDate(filing?.date)}

Gelir: ${money(metrics.revenue?.current)}
Net kâr/zarar: ${money(metrics.netIncome?.current)}
EPS: ${number(metrics.eps?.current)}
Nakit: ${money(metrics.cash?.current)}
Operasyonel nakit akışı: ${money(metrics.operatingCashFlow?.current)}
Toplam varlıklar: ${money(metrics.assets?.current)}
Toplam yükümlülükler: ${money(metrics.liabilities?.current)}

Sky bilanço değerlendirmesi:
${comments.length
  ? comments.map((x) => `• ${x}`).join('\n')
  : '• Karşılaştırma için yeterli standart SEC kalemi bulunamadı.'}

Bilanço puanı: ${score}
SONUÇ: ${verdict}

Not: Karşılaştırma SEC XBRL'deki son iki raporlanan veri noktası üzerinden yapılır. Şirketlerin raporlama dönemleri farklı olabileceğinden bu sonuç yatırım tavsiyesi değildir.`;

  return {
    symbol,
    verdict,
    score,
    filing,
    metrics,
    answer,
  };
}

function growth(metric) {
  const current = metric?.current;
  const previous = metric?.previous;

  if (
    !Number.isFinite(current) ||
    !Number.isFinite(previous) ||
    previous === 0
  ) {
    return null;
  }

  return ((current - previous) / Math.abs(previous)) * 100;
}

function money(value) {
  if (!Number.isFinite(value)) return '—';

  const abs = Math.abs(value);

  if (abs >= 1_000_000_000) {
    return `$${(value / 1_000_000_000).toFixed(2)} milyar`;
  }

  if (abs >= 1_000_000) {
    return `$${(value / 1_000_000).toFixed(2)} milyon`;
  }

  return `$${value.toLocaleString('tr-TR')}`;
}

function number(value) {
  return Number.isFinite(value)
    ? Number(value).toLocaleString('tr-TR', {
        maximumFractionDigits: 3,
      })
    : '—';
}

function formatDate(value) {
  if (!value) return '—';

  return new Date(`${value}T12:00:00Z`).toLocaleDateString(
    'tr-TR',
    {
      day: '2-digit',
      month: 'long',
      year: 'numeric',
      timeZone: 'Europe/Istanbul',
    }
  );
}
